% Abstract environment - from print styles (8pt font)
\newenvironment{articleabstract}[1][en]{%
  \fontsize{8}{12.554}\selectfont\color{darktext}
  \def\abstractlang{#1}%
  \ifx\abstractlang\frenchlang
    \begin{otherlanguage}{french}
  \fi
}{%
  \ifx\abstractlang\frenchlang
    \end{otherlanguage}
  \fi
  \normalsize\color{black}
}

% Helper command for language comparison
\def\frenchlang{fr}

% Two-column abstract layout
\newcommand{\abstractlayout}[2]{%
  \begin{multicols}{2}
    \begin{articleabstract}[en]
      #1
    \end{articleabstract}
    \columnbreak
    \begin{articleabstract}[fr]
      #2
    \end{articleabstract}
  \end{multicols}
  \vspace{12pt}
}
